# 1. Add 'contrib' to your package sources
sudo sed -i 's/main$/main contrib/' /etc/apt/sources.list

# 2. Update package lists
sudo apt-get update

# 3. Now install guest additions
sudo apt-get install -y virtualbox-guest-utils

# 4. Add your user to vboxsf group
sudo usermod -aG vboxsf vboxuser

# 5. Reboot
sudo reboot
